4// A allowlist de routers, compilada.
5//
6// Para cada rede e provedor: o `to` aceito, o spender que recebe o approve exato, as
7// funcoes aceitas (seletores tirados do ABI verificado) e como conferir que o codigo
8// atras do endereco ainda e o que foi auditado. A allowlist vive no binario: a config
9// remota so pode desligar provedor, nunca acrescentar endereco (docs/blockchain.md 3.5).
10//
11// Conferencia feita em 25/09/2026, para cada endereco e cada uma das 7 redes:
12// - `eth_getCode` nao vazio em dois RPCs;
13// - codigo verificado no Sourcify (sourcify.dev/server/v2/contract/<chainId>/<endereco>):
14// AugustusV6 (solc 0.8.22), MetaAggregationRouterV2, GenericSwapFacetV3,
15// OpenOceanExchange (implementacao atras do proxy);
16// - o endereco devolvido pela API ao vivo (`to` da transacao e `approvalAddress`/
17// `tokenTransferProxy`/`routerAddress`) bateu com o daqui.
18//
19// Segunda leva, conferida em 26/09/2026 do mesmo jeito (codigo em dois RPCs, Sourcify ou
20// explorador, `to` da API ao vivo): Plasma, Linea, Unichain e Sonic. Cada provedor so
21// entra onde tudo fechou; o que ficou de fora esta dito na lista do provedor.
22
23/// Como conferir, antes de assinar, que o codigo atras do router e o que foi lido.
24public enum TradeRouterPin: Sendable, Equatable {
25 /// Contrato sem proxy: o codigo e imutavel. A Augustus V6.2 e um diamond, mas
26 /// `swapExactAmountIn` esta no bytecode principal (Routers.sol), e funcao declarada
27 /// tem precedencia sobre o fallback de facetas.
28 case immutable
29 /// Proxy EIP-1967: a implementacao no slot
30 /// `0x360894a13ba1a3210667c828492db98dca3e2076cc3735a920a3ca505d382bbc` tem de ser
31 /// esta. Upgrade do dono do proxy desliga o provedor ate a proxima versao do app.
32 case eip1967(implementation: EVMAddress)
33 /// Diamond EIP-2535: `facetAddress(seletor)` tem de devolver esta faceta para cada
34 /// seletor aceito.
35 case diamondFacet(EVMAddress)
36
37 /// `keccak256("eip1967.proxy.implementation") - 1`.
38 public static let eip1967ImplementationSlot = [UInt8](hex: "360894a13ba1a3210667c828492db98dca3e2076cc3735a920a3ca505d382bbc")!
39
40 /// `facetAddress(bytes4)` do DiamondLoupe (EIP-2535).
41 public static let facetAddressFunction = try! ABIFunction("facetAddress(bytes4)") // cdffacc6
42
43 /// O endereco que a leitura na cadeia tem de devolver; `nil` se nao ha o que ler.
44 public var expected: EVMAddress? {
45 switch self {
46 case .immutable: return nil
47 case .eip1967(let implementation): return implementation
48 case .diamondFacet(let facet): return facet
49 }
50 }
51}

53/// Um router aceito numa rede.
54public struct TradeRouter: Sendable, Equatable {
55 public let provider: TradeProvider
56 public let chainID: UInt64
57 /// O `to` da transacao de troca.
58 public let address: EVMAddress
59 /// Quem recebe o `approve` exato do token vendido.
60 public let spender: EVMAddress
61 public let pin: TradeRouterPin
62 /// As funcoes aceitas. Seletor fora desta lista e recusado antes de decodificar.
63 public let functions: [ABIFunction]
64 /// LI.FI: o contrato de taxa que pode aparecer dentro do `swapData`.
65 public let feeForwarder: EVMAddress?
66
67 public var selectors: Set<[UInt8]> { Set(functions.map(\.selector)) }
68
69 /// O router puxa o token pelo Permit2 (Uniswap): o approve exato vai ao Permit2, e o
70 /// plano soma a permissao do Permit2 a este router (`EVMPermit2Approval`).
71 public var viaPermit2: Bool { spender == EIP712ValidatedMessage.permit2 }
72
73 /// A calldata de `facetAddress(seletor)` para cada seletor aceito (so diamond).
74 public var facetQueries: [[UInt8]] {
75 guard case .diamondFacet = pin else { return [] }
76 return functions.compactMap { function in
77 try? TradeRouterPin.facetAddressFunction.encodeCall([.fixedBytes(function.selector)])
78 }
79 }
80}

166 /// O router de um provedor numa rede, ou `nil` se o provedor nao atende ali.
167 public static func router(for provider: TradeProvider, on chain: Chain) -> TradeRouter? {
168 guard chain.family == .evm, let chainID = chain.evmChainID, evmChainIDs.contains(chainID) else { return nil }
169 switch provider {
170 case .velora:
171 guard veloraChainIDs.contains(chainID) else { return nil }
172 return TradeRouter(provider: .velora, chainID: chainID, address: velora, spender: velora,
173 pin: .immutable, functions: VeloraCalldata.functions, feeForwarder: nil)
174 case .kyberSwap:
175 guard kyberChainIDs.contains(chainID) else { return nil }
176 return TradeRouter(provider: .kyberSwap, chainID: chainID, address: kyber, spender: kyber,
177 pin: .immutable, functions: KyberCalldata.functions, feeForwarder: nil)
178 case .lifi:
179 guard let facet = lifiFacets[chainID] else { return nil }
180 return TradeRouter(provider: .lifi, chainID: chainID, address: lifiDiamond, spender: lifiDiamond,
181 pin: .diamondFacet(facet), functions: LiFiCalldata.functions, feeForwarder: lifiFeeForwarder)
182 case .de1:
183 guard let implementation = de1Implementations[chainID] else { return nil }
184 return TradeRouter(provider: .de1, chainID: chainID, address: de1Proxy, spender: de1Proxy,
185 pin: .eip1967(implementation: implementation), functions: De1Calldata.functions, feeForwarder: nil)
186 case .uniswap:
187 // O token vendido e aprovado ao Permit2, e o Permit2 da ao router a permissao
188 // exata de 30 minutos (`TradeRouter.viaPermit2`). Router sem proxy.
189 guard let deployment = UniswapDeployment.all[chainID] else { return nil }
190 return TradeRouter(provider: .uniswap, chainID: chainID, address: deployment.router, spender: EIP712ValidatedMessage.permit2,
191 pin: .immutable, functions: UniswapCalldata.functions, feeForwarder: nil)
192 case .pancakeswap:
193 // SmartRouter sem proxy, com o approve exato a ele mesmo. O `Ownable` dele so
194 // mexe na rota de stable swap, que a carteira nao monta.
195 guard let deployment = PancakeDeployment.all[chainID] else { return nil }
196 return TradeRouter(provider: .pancakeswap, chainID: chainID, address: deployment.router, spender: deployment.router,
197 pin: .immutable, functions: PancakeCalldata.functions, feeForwarder: nil)
198 case .shibaswap:
199 // UniswapV2Router02 sem proxy e sem dono, com o approve exato a ele mesmo.
200 guard let deployment = ShibaSwapDeployment.all[chainID] else { return nil }
201 return TradeRouter(provider: .shibaswap, chainID: chainID, address: deployment.router, spender: deployment.router,
202 pin: .immutable, functions: ShibaSwapCalldata.functions, feeForwarder: nil)
203 }
204 }

206 /// Todos os routers aceitos numa rede.
207 public static func routers(on chain: Chain) -> [TradeRouter] {
208 TradeProvider.allCases.compactMap { router(for: $0, on: chain) }
209 }
210
211 /// Os spenders de troca de uma rede, para a tela de aprovacoes vigentes e revogacao
212 /// (docs/seguranca.md 4.4), incluindo o VaultRelayer da CoW.
213 public static func spenders(on chain: Chain) -> Set<EVMAddress> {
214 var out = Set(routers(on: chain).map(\.spender))
215 if CoWProtocol.supports(chain) { out.insert(CoWProtocol.vaultRelayer) }
216 return out
217 }
