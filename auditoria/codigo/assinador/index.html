5/// O unico lugar do app onde uma chave privada assina alguma coisa.
6///
7/// Recebe um `SigningPlan` (que so existe se passou pela validacao da rede) e a RK
8/// recem-destrancada por presenca do dono. Sincrono de ponta a ponta, sem `await`:
9/// funcao assincrona guarda variaveis locais num contexto no heap que e liberado sem
10/// zerar, e aqui as variaveis locais sao a seed e as chaves (docs/seguranca.md §2.9).
11///
12/// ```
13/// rk ─► DEK ─► registro ─► frase ─► seed ─► chave de cada pedido ─► assinatura
14/// cada etapa zera a anterior; a seed e as chaves somem no fim do plano
15/// ```
16public enum Signer {
17 public enum Failure: Error, Equatable, Sendable {
18 case expired
19 /// A chave derivada nao e a que a transacao espera: caminho errado ou
20 /// carteira errada. Nada foi assinado.
21 case unexpectedKey
22 /// A assinatura produzida nao verificou. Com nonce deterministico, uma
23 /// assinatura defeituosa pode entregar a chave; ela nunca sai daqui.
24 case verificationFailed
25 case unsupportedScheme(SignatureScheme)
26 /// A passphrase digitada nao da a carteira (o fingerprint da raiz nao bate). Nada foi
27 /// assinado.
28 case passphraseMismatch
29 }
30
31 /// Assina todas as transacoes do plano, na ordem. Consome e zera a RK.
32 ///
33 /// `passphrase`: numa carteira com passphrase, a que o dono acabou de digitar (ou a que a
34 /// folga de alguns minutos guarda na memoria). O cofre nao tem passphrase nenhuma (pedido do
35 /// dono, 05/11/2026): a seed nasce da entropia do cofre com esta, em NFKD, e a raiz tem de dar
36 /// o `fingerprint` que a criacao gravou, antes de nascer qualquer chave; nao da, nada e
37 /// assinado. Com o fingerprint, a raiz confere tambem sem passphrase. O chamador continua dono do buffer. Sem valor padrao: cada chamador diz se ha
38 /// passphrase, e um chamador novo nao fura a regra esquecendo o parametro. Sem ela, uma
39 /// carteira com passphrase daria as chaves da carteira de base, que a `expectedPublicKey`
40 /// recusa.
41 public static func sign(_ plan: SigningPlan, rootKey rk: SecureBytes, vault: WalletVault,
42 passphrase typed: SecureBytes?, fingerprint: [UInt8]?) throws -> [SignedTransaction] {
43 // Por ultimo: a pilha que as contas da BoringSSL deixaram (StackWipe).
44 defer { StackWipe.below() }
45 defer { rk.wipe() }
46 guard !plan.isExpired() else { throw Failure.expired }
47 // O nonce conferido antes de abrir o cofre. A derivacao aqui nao passa pelo `DerivationSelfTest`, de proposito: a
48 // chave tem de bater com a `expectedPublicKey` e a assinatura e verificada, e uma derivacao errada so falha fechado.
49 // Recusar o envio por ela congelaria o saldo sem proteger nada (revisao da decima terceira passada).
50 guard SigningSelfTest.passed(for: plan) else { throw SigningSelfTest.Failure() }
51
52 let secret = PassphraseProof.joining(try vault.open(walletID: plan.walletID, rk: rk), typed)
53 rk.wipe()
54 defer { secret.wipe() }
55 let seed = try secret.seed()
56 defer { seed.wipe() }
57
58 var masters: [Curve: HDKey] = [:]
59 defer { masters.values.forEach { $0.wipe() } }
60 func master(_ curve: Curve) throws -> HDKey {
61 if let cached = masters[curve] { return cached }
62 let made = try HDKey.master(seed: seed, curve: curve)
63 masters[curve] = made
64 return made
65 }
66 // A raiz confere antes de qualquer chave de pedido, com ou sem passphrase: a carteira com passphrase que chegasse
67 // sem ela daria a raiz da carteira de base, e a recusa vem aqui, e nao na chave de cada pedido.
68 if let fingerprint {
69 guard Hash.constantTimeEqual(try master(.secp256k1).fingerprint(), fingerprint) else {
70 throw typed == nil ? Failure.unexpectedKey : Failure.passphraseMismatch
71 }
72 } else if typed != nil {
73 throw Failure.passphraseMismatch
74 }
75 // Cardano: a mestra Icarus sai da entropia (CIP-3), antes de o segredo ser zerado. A passphrase com acento que uma
76 // versao anterior gravou fora de NFKD deu a conta da Cardano pelos bytes digitados: a mestra deles fica de reserva,
77 // para essa carteira mover o saldo da Cardano como a tela de aviso pede.
78 let cardano = try CardanoSigning.master(for: plan, secret: secret)
79 defer { cardano?.wipe() }
80 let legacyCardano = try typed.flatMap { try PassphraseProof.legacyCardanoMaster(for: plan, stored: secret, typed: $0) }
81 defer { legacyCardano?.wipe() }
82 secret.wipe()
83
84 var signed: [SignedTransaction] = []
85 for transaction in plan.transactions {
86 var signatures: [ProducedSignature] = []
87 for request in transaction.signingRequests {
88 if request.scheme == .ed25519Cardano {
89 do {
90 signatures.append(try CardanoSigning.sign(request, master: cardano))
91 } catch Failure.unexpectedKey where legacyCardano != nil {
92 signatures.append(try CardanoSigning.sign(request, master: legacyCardano))
93 }
94 continue
95 }
96 let key = try master(request.curve).derive(request.path)
97 defer { key.wipe() }
98 signatures.append(try sign(request, with: key))
99 }
100 signed.append(try transaction.assemble(with: signatures))
101 }
102 return signed
103 }

105 static func sign(_ request: SigningRequest, with key: HDKey) throws -> ProducedSignature {
106 let publicKey = try key.publicKey()
107 guard Hash.constantTimeEqual(publicKey, request.expectedPublicKey) else { throw Failure.unexpectedKey }
108
109 switch request.scheme {
110 case .ecdsaDER:
111 let der = try Secp256k1.signDER(digest: request.payload, privateKey: key.key, hedged: true)
112 guard Secp256k1.verifyDER(signature: der, digest: request.payload, publicKey: publicKey) else {
113 throw Failure.verificationFailed
114 }
115 return ProducedSignature(bytes: der)
116 case .ecdsaRecoverable:
117 let (compact, recovery) = try Secp256k1.signRecoverable(digest: request.payload, privateKey: key.key, hedged: true)
118 let recovered = try Secp256k1.recover(digest: request.payload, compact: compact, recoveryID: recovery, compressed: true)
119 guard recovered == publicKey else { throw Failure.verificationFailed }
120 return ProducedSignature(bytes: compact, recoveryID: recovery)
121 case .ed25519:
122 let signature = try Ed25519.sign(request.payload, seed: key.key, hedged: true)
123 guard Ed25519.verify(signature: signature, message: request.payload, publicKey: publicKey) else {
124 throw Failure.verificationFailed
125 }
126 return ProducedSignature(bytes: signature)
127 case .schnorrBIP340:
128 throw Failure.unsupportedScheme(.schnorrBIP340)
129 case .ed25519Cardano:
130 // Chave BIP32-Ed25519, que o HDKey nao deriva: so pelo CardanoSigning.
131 throw Failure.unsupportedScheme(.ed25519Cardano)
132 }
133 }
134}

136/// A passphrase de uma carteira com passphrase nunca e gravada (pedido do dono, 05/11/2026). O cofre guarda
137/// so a entropia e o idioma; a passphrase entra digitada, so durante a operacao, e confere pelo fingerprint
138/// BIP-32 da raiz que a criacao gravou nos metadados (4 bytes do hash da chave publica, que nao dizem a
139/// passphrase). Quem tem o PIN e as palavras ja podia testar passphrases fora do aparelho pelos enderecos;
140/// o fingerprint nao abre nada a mais.
141public enum PassphraseProof {
142 public enum Result: Sendable, Equatable {
143 case match
144 case mismatch
145 /// Confere, mas uma versao anterior gravou a passphrase fora de NFKD, e a conta da Cardano dela saiu
146 /// dos bytes digitados: restaurada pelas palavras, ela muda.
147 case legacy
148 }

150 /// O segredo de uso: a entropia do cofre (que passa a ser dele) e uma copia da digitada em NFKD. O
151 /// chamador continua dono de `typed`.
152 static func joining(_ stored: WalletSecret, _ typed: SecureBytes?) -> WalletSecret {
153 stored.passphrase.wipe()
154 return WalletSecret(entropy: stored.entropy, language: stored.language, passphrase: typed.map(BIP39.nfkd) ?? SecureBytes(capacity: 1))
155 }
156
157 /// A passphrase digitada da a carteira? Pelo cofre, com a RK aberta agora (consumida). `cardanoPublicKey`:
158 /// a chave da conta Cardano gravada, para achar a passphrase de uma versao anterior (`legacy`).
159 public static func check(walletID: UUID, rootKey rk: SecureBytes, vault: WalletVault, typed: SecureBytes,
160 fingerprint: [UInt8]?, cardanoPublicKey: [UInt8]?) throws -> Result {
161 defer { rk.wipe() }
162 let stored = try vault.open(walletID: walletID, rk: rk)
163 rk.wipe()
164 defer { stored.wipe() }
165 return try check(stored, typed: typed, fingerprint: fingerprint, cardanoPublicKey: cardanoPublicKey)
166 }
167
168 /// Com o registro ja aberto (que continua de quem chama).
169 public static func check(_ stored: WalletSecret, typed: SecureBytes, fingerprint: [UInt8]?, cardanoPublicKey: [UInt8]?) throws -> Result {
170 let entropy = SecureBytes(capacity: stored.entropy.count)
171 stored.entropy.withUnsafeBytes { entropy.append(contentsOf: $0.bindMemory(to: UInt8.self)) }
172 let secret = WalletSecret(entropy: entropy, language: stored.language, passphrase: BIP39.nfkd(typed))
173 defer { secret.wipe() }
174 return try verify(secret, fingerprint: fingerprint, cardanoPublicKey: cardanoPublicKey)
175 }
176
177 /// Pela frase em buffer (as palavras que a tela acabou de mostrar), sem o cofre.
178 public static func check(phrase: SecureBytes, language: BIP39Language, typed: SecureBytes,
179 fingerprint: [UInt8]?, cardanoPublicKey: [UInt8]?) throws -> Result {
180 let stored = try WalletSecret.from(phrase: phrase, language: language)
181 let secret = joining(stored, typed)
182 defer { secret.wipe() }
183 return try verify(secret, fingerprint: fingerprint, cardanoPublicKey: cardanoPublicKey)
184 }
185
186 /// A raiz tem o fingerprint da criacao? E, com a conta Cardano gravada, ela e a desta passphrase em NFKD? A
187 /// passphrase certa com outra conta Cardano e a que uma versao anterior gravou fora de NFKD.
188 static func verify(_ secret: WalletSecret, fingerprint: [UInt8]?, cardanoPublicKey: [UInt8]?) throws -> Result {
189 defer { StackWipe.below() }
190 guard let fingerprint, secret.passphrase.count > 0 else { return .mismatch }
191 let seed = try secret.seed()
192 defer { seed.wipe() }
193 let root = try HDKey.master(seed: seed, curve: .secp256k1)
194 defer { root.wipe() }
195 guard Hash.constantTimeEqual(try root.fingerprint(), fingerprint) else { return .mismatch }
196 guard let cardanoPublicKey else { return .match }
197 return try CardanoKey.derivedAccount(secret).publicKey == cardanoPublicKey ? .match : .legacy
198 }

200 /// A mestra Icarus pelos bytes digitados, so quando o plano tem Cardano e o NFKD muda a passphrase.
201 static func legacyCardanoMaster(for plan: SigningPlan, stored secret: WalletSecret, typed: SecureBytes) throws -> CardanoKey? {
202 guard !Hash.constantTimeEqual(secret.passphrase, typed) else { return nil }
203 let raw = rawSecret(secret, typed)
204 defer { raw.wipe() }
205 return try CardanoSigning.master(for: plan, secret: raw)
206 }

208 private static func rawSecret(_ secret: WalletSecret, _ typed: SecureBytes) -> WalletSecret {
209 let entropy = SecureBytes(capacity: secret.entropy.count)
210 secret.entropy.withUnsafeBytes { entropy.append(contentsOf: $0.bindMemory(to: UInt8.self)) }
211 let passphrase = SecureBytes(capacity: max(typed.count, 1))
212 typed.withUnsafeBytes { passphrase.append(contentsOf: $0.bindMemory(to: UInt8.self)) }
213 return WalletSecret(entropy: entropy, language: secret.language, passphrase: passphrase)
214 }

216 /// Os enderecos das redes que faltam numa carteira com passphrase, com a passphrase digitada agora, depois de
217 /// ela conferir: uma rede que entrou numa versao nova so pode nascer assim, porque o cofre nao tem a
218 /// passphrase. Nao consome a RK: quem chama assina logo depois com ela. Nao confere, nada.
219 public static func missingAccounts(walletID: UUID, rootKey rk: SecureBytes, vault: WalletVault, typed: SecureBytes,
220 fingerprint: [UInt8]?, chains: [Chain]) throws -> [DerivedAccount] {
221 guard !chains.isEmpty else { return [] }
222 let secret = joining(try vault.open(walletID: walletID, rk: rk), typed)
223 defer { secret.wipe() }
224 guard try verify(secret, fingerprint: fingerprint, cardanoPublicKey: nil) == .match else { return [] }
225 return try AccountDeriver.derive(secret, chains: chains).accounts
226 }

229public enum AccountDeriver {
230 /// Deriva o endereco padrao de cada rede, depois de o aparelho conferir os vetores publicos (`DerivationSelfTest`): com
231 /// um resultado diferente, nenhuma carteira e derivada (decima terceira passada da auditoria de 03/11/2026).
232 public static func derive(_ secret: WalletSecret, chains: [Chain] = Chain.all) throws -> (accounts: [DerivedAccount], fingerprint: [UInt8]) {
233 defer { StackWipe.below() }
234 guard DerivationSelfTest.passed else { throw DerivationSelfTest.Failure() }
235 // Duas vezes, iguais: um defeito intermitente do aparelho (memoria, otimizador) que tivesse passado no autoteste nao
236 // grava um endereco que a frase nao reabre (revisao da decima terceira passada).
237 let first = try deriveUnchecked(secret, chains: chains)
238 let second = try deriveUnchecked(secret, chains: chains)
239 guard first.accounts == second.accounts, first.fingerprint == second.fingerprint else { throw DerivationSelfTest.Failure() }
240 return first
241 }
242
243 /// A derivacao em si, sem a conferencia: so para o proprio `DerivationSelfTest`.
244 static func deriveUnchecked(_ secret: WalletSecret, chains: [Chain]) throws -> (accounts: [DerivedAccount], fingerprint: [UInt8]) {
245 let seed = try secret.seed()
246 defer { seed.wipe() }
247 let secp = try HDKey.master(seed: seed, curve: .secp256k1)
248 let ed = try HDKey.master(seed: seed, curve: .ed25519)
249 defer { secp.wipe(); ed.wipe() }
250
251 var accounts: [DerivedAccount] = []
252 for chain in chains {
253 if chain.family == .cardano {
254 if let account = try? CardanoKey.derivedAccount(secret, chain: chain) { accounts.append(account) }
255 continue
256 }
257 let path = DefaultPaths.path(for: chain)
258 let master = chain.family.curve == .secp256k1 ? secp : ed
259 let key = try master.derive(path)
260 defer { key.wipe() }
261 let publicKey = try key.publicKey()
262 guard let address = try? Address.from(publicKey: publicKey, chain: chain) else { continue }
263 var xpub: ExtendedPublicKey?
264 if chain.family == .utxo {
265 let accountPath = DerivationPath(components: Array(path.components.prefix(3)))
266 let account = try master.derive(accountPath)
267 defer { account.wipe() }
268 xpub = try ExtendedPublicKey(publicKey: account.publicKey(), chainCode: account.chainCode.withUnsafeBytes { Array($0) })
269 }
270 accounts.append(DerivedAccount(chainID: chain.id, path: path, address: address, publicKey: publicKey, accountXPub: xpub))
271 }
272 return (accounts, try secp.fingerprint())
273 }
