12/// A chave raiz (RK) e os dois caminhos que a destrancam: PIN e biometria.
13///
14/// ```
15/// PIN ─Argon2id─► AP ─┬─(senha de aplicativo do item)─► item rk.pin
16/// ├─HKDF─► senha da chave do PIN no SE
17/// └─HKDF─► K_pin
18/// item rk.pin = embrulho(chave do PIN no SE, ChaChaPoly(K_pin, RK)) ─SE+senha─► ─K_pin─► RK
19/// Face ID (decisao no SEP) ─► item rk.bio = embrulho(K_bio, RK) ─SE─► RK
20/// ```
21///
22/// Formato 2 do slot (auditoria 3, A1): a chave do SE que embrulha o slot tem senha de
23/// aplicativo derivada do PIN, entao o SE so desembrulha com o PIN certo. Quem tirar uma
24/// copia do slot nao consegue mais testar PINs fora do aparelho depois de uma unica
25/// chamada ao SE: cada chute passa pelo SE deste iPhone, mais um Argon2id. O formato 1
26/// (chave do aparelho sem senha) migra no primeiro PIN certo.
27///
28/// **Nao existe verificador de PIN gravado em lugar nenhum.** O PIN errado e recusado
29/// pelo chaveiro, que nao devolve o item sem a senha de aplicativo certa, e de novo
30/// pela camada interna, que so abre com a chave derivada do PIN. A camada interna e o
31/// que garante o desenho mesmo se a senha de aplicativo for so uma regra de acesso e
32/// nao entrar na chave do item: quem copiar o item e conseguir usar o SE deste aparelho
33/// ainda precisa do PIN, e cada chute custa um Argon2id de 256 MiB
34/// (docs/seguranca.md §2).

226 /// Destranca pelo PIN. O incremento de tentativa e gravado **antes** da avaliacao;
227 /// se a gravacao falhar, a tentativa nao e avaliada (falha fechada).
228 ///
229 /// Com uma troca de PIN interrompida existem dois slots, e os dois sao tentados na
230 /// mesma tentativa. O que abrir decide o desfecho: o antigo desfaz a troca, o novo
231 /// a conclui.

237 private func unlockLocked(pin: SecureBytes) throws -> SecureBytes {
238 switch securityMode {
239 case .hardened: throw Failure.hardenedModeActive
240 case .unknown: throw Failure.storage
241 case .normal: break
242 }
243 let pendingAccount = Account.rootByPIN + Account.pendingSuffix
244 let main = store.probe(Account.rootByPIN)
245 let pending = store.probe(pendingAccount)
246 guard main != .unknown, pending != .unknown else { throw Failure.storage }
247 guard main == .present || pending == .present else { throw Failure.notSetUp }
248
249 // PIN sem forma (envio duplo com o buffer vazio) nunca gasta tentativa.
250 guard Self.isWellFormed(pin) else { throw Failure.malformedPIN }
251 let record = try rebasedAttemptsLocked()
252 let waiting = record.remaining()
253 guard waiting <= 0 else { throw Failure.throttled(seconds: waiting) }
254
255 let next = AttemptRecord.after(failures: record.failures + 1)
256 do { try store.update(next.encoded, account: Account.attempts) } catch { throw Failure.storage }
257
258 let candidates = [main == .present ? "" : nil, pending == .present ? Account.pendingSuffix : nil].compactMap { $0 }
259 var opened: (rk: SecureBytes, suffix: String)?
260 for suffix in candidates where opened == nil {
261 do {
262 opened = (try readPINSlot(pin: pin, suffix: suffix), suffix)
263 } catch Failure.wrongPIN {
264 continue
265 }
266 }
267
268 guard let opened else {
269 if let threshold = wipeThreshold, next.failures >= threshold {
270 wipeAllLocked()
271 throw Failure.wiped
272 }
273 let left = wipeThreshold.map { $0 - next.failures }
274 throw Failure.wrongPIN(remainingBeforeWipe: left)
275 }
276
277 // Contador de volta a zero. Se falhar, o dono so espera mais na proxima vez.
278 try? store.update(AttemptRecord.zero.encoded, account: Account.attempts)
279 recordPINThisBoot()
280 // Um rk.and ao lado do rk.pin e uma transicao que nao chegou ao fim (ligar
281 // interrompido antes do commit, ou desligar interrompido depois do rk.pin gravado).
282 // O PIN acabou de provar o slot do modo normal: o rk.and sai.
283 if store.probe(Account.rootHardened) == .present { discardHardenedLocked() }
284 if pending == .present {
285 if opened.suffix.isEmpty {
286 deleteSlot(suffix: Account.pendingSuffix)
287 } else {
288 try? promotePendingLocked(rk: opened.rk, pin: pin)
289 }
290 }
291 // Slot no formato 1: passa para o 2 agora que o PIN certo esta na mao. Falha aqui
292 // nao tranca ninguem: antes do ponto sem volta o slot antigo continua o unico.
293 if slotVersion(suffix: "") == 1 {
294 try? migrateLocked(rk: opened.rk, pin: pin)
295 } else {
296 dropLegacyKeyIfDone()
297 }
298 return opened.rk
299 }

413 /// A1: o que entrega ou destroi a carteira inteira (ver as palavras, lacrar o
414 /// envelope, remover, desligar o Face ID) pede o PIN e, com o Face ID ligado, o rosto
415 /// tambem. Quem viu o PIN e pegou o iPhone nao leva a frase; quem apresenta o rosto do
416 /// dono a forca tambem nao. Sem o Face ID ligado, so o PIN.
417 ///
418 /// O PIN vem primeiro e conta tentativa como em `unlock`; o rosto e decidido no SEP,
419 /// pelo slot da biometria, e as duas chaves raiz tem de ser a mesma. Nenhum dos dois
420 /// caminhos cai para o codigo do iPhone.
421 ///
422 /// O Face ID que cai sem o dono pedir (cadastro de rostos mudou, Face ID tirado do app
423 /// ou do iPhone) apaga o slot e comeca a espera de `PINPolicy.biometryLossHold`:
424 /// durante ela, isto recusa antes de pedir qualquer coisa, mesmo com um Face ID
425 /// religado. Sem a espera, o ladrao que viu o codigo do iPhone cadastrava o proprio
426 /// rosto e passava com o PIN so. Chaveiro mudo recusa: nunca vira "Face ID desligado".

433 try checkPINAndBiometryAllowed()

435 guard slot != .unknown else { throw Failure.storage }
436
437 let rk = try unlockLocked(pin: pin)
438 guard slot == .present else { return rk }

441 // (revisao do A1: uma falha passageira, ou o Espelhamento, derrubava o slot).
442 if biometryAvailability() == .gone { throw Failure.biometryUnavailable }

444 defer { check.wipe() }
445 guard Hash.constantTimeEqual(check, rk) else { throw Failure.storage }
446 return rk
447 } catch {
448 rk.wipe()
449 throw error
450 }

494 /// O Face ID caiu sem o dono pedir: a espera e a marca de perdido sao gravadas
495 /// primeiro, e so entao o slot sai. Se uma delas nao gravar, o slot fica: o rosto
496 /// continua sem abrir, e o que pede PIN e rosto continua travado, em vez de passar com

690 /// Troca atomica. O slot novo e gravado com sufixo e conferido; o ponto sem volta
691 /// e apagar o slot antigo. Antes dele, qualquer falha desfaz a troca e o PIN antigo
692 /// continua o unico. Depois dele, a troca esta feita: o slot pendente abre com o
693 /// PIN novo, e se a promocao a principal falhar aqui, o proximo desbloqueio a
694 /// conclui. Em nenhum instante o dono fica sem um slot que abre.
695 public func changePIN(rk: SecureBytes, newPIN: SecureBytes) throws {
696 lock.lock(); defer { lock.unlock() }
697 try replacePINLocked(rk: rk, newPIN: newPIN)
698 dropLegacyKeyIfDone()
699 }

701 private func replacePINLocked(rk: SecureBytes, newPIN: SecureBytes) throws {
702 guard Self.isWellFormed(newPIN) else { throw Failure.malformedPIN }
703 let pendingAccount = Account.rootByPIN + Account.pendingSuffix
704 // Uma troca interrompida se resolve no proximo desbloqueio, nao aqui.
705 guard store.probe(Account.rootByPIN) == .present else { throw Failure.storage }
706 switch store.probe(pendingAccount) {
707 case .unknown: throw Failure.storage
708 case .present: deleteSlot(suffix: Account.pendingSuffix)
709 case .absent: break
710 }
711
712 do {
713 try writePINSlot(rk: rk, pin: newPIN, kdf: pinParameters, suffix: Account.pendingSuffix)
714 try confirm(pin: newPIN, suffix: Account.pendingSuffix, matches: rk)
715 try store.delete(Account.rootByPIN)
716 } catch {
717 deleteSlot(suffix: Account.pendingSuffix)
718 throw error is Failure ? error : Failure.storage
719 }
720 // Ponto sem volta passado: o PIN antigo nao abre mais nada.
721 try? store.delete(Account.pinKDF)
722 try? store.delete(Account.pinLength)
723 try? promotePendingLocked(rk: rk, pin: newPIN)
724 }

726 /// Regrava o slot principal com o PIN que abriu o pendente, confere, e so entao
727 /// apaga o pendente.
728 private func promotePendingLocked(rk: SecureBytes, pin: SecureBytes) throws {
729 if store.probe(Account.rootByPIN) == .present {
730 try store.delete(Account.rootByPIN)
731 }
732 try writePINSlot(rk: rk, pin: pin, kdf: pinParameters, suffix: "")
733 try confirm(pin: pin, suffix: "", matches: rk)
734 deleteSlot(suffix: Account.pendingSuffix)
735 }

739 try? store.delete(Account.pinKDF + suffix)
740 try? store.delete(Account.pinLength + suffix)

742 // gravacao de um principal novo a substitui.

824 /// Grava um slot do PIN. Pre-condicao: o slot nao existe. O registro de custo e
825 /// gravado antes e desfeito se o slot falhar, para nunca sobrar um slot sem custo.
826 func writePINSlot(rk: SecureBytes, pin: SecureBytes, kdf: KDFParameters, suffix: String) throws {
827 guard store.probe(Account.rootByPIN + suffix) == .absent else { throw Failure.storage }
828 let saltBytes = try SecureBytes.random(count: VaultFormat.saltLength)
829 let salt = saltBytes.withUnsafeBytes { Array($0) }
830 let ap = try KeyDerivation.deriveMasterKey(password: pin, salt: salt, parameters: kdf)
831 defer { ap.wipe() }
832
833 let inner = try Self.sealInner(rk, ap: ap, salt: salt)
834 defer { inner.wipe() }
835 // Formato 2: a chave do SE deste slot so desembrulha com a senha derivada do PIN.
836 let slot = Self.wrapSlot(suffix)
837 let password = Self.devicePassword(ap: ap, salt: salt)
838 defer { password.wipe() }
839 try password.withUnsafeData { try wrapper.createKey(slot, password: $0) }
840 let blob: Data
841 do {
842 blob = try password.withUnsafeData { try wrapper.wrap(inner, slot: slot, password: $0) }
843 } catch {
844 wrapper.deleteKey(slot)
845 throw error
846 }
847
848 var kdfRecord = salt
849 kdfRecord += kdf.memoryKiB.bigEndianByteArray + kdf.passes.bigEndianByteArray + [kdf.lanes]
850 kdfRecord.append(SlotFormat.version)
851 try? store.delete(Account.pinKDF + suffix)
852 try? store.delete(Account.pinLength + suffix)
853 do {
854 // O tamanho vai antes do slot e sai junto se ele falhar: nunca existe um slot
855 // de 4 digitos que a tela trate como de 6.
856 try store.add(Data([UInt8(pin.count)]), account: Account.pinLength + suffix, protection: .standard, context: nil)
857 try store.add(Data(kdfRecord), account: Account.pinKDF + suffix, protection: .standard, context: nil)
858 let context = try store.applicationPasswordContext(ap)
859 try store.add(blob, account: Account.rootByPIN + suffix, protection: .applicationPassword, context: context)
860 } catch {
861 try? store.delete(Account.pinKDF + suffix)
862 try? store.delete(Account.pinLength + suffix)
863 wrapper.deleteKey(slot)
864 throw error
865 }
866 }

890 record.count == SlotFormat.legacyLength || (record.count == SlotFormat.length && record.last == SlotFormat.version)
891 else {
892 throw Failure.storage

898 let passes = bytes[20..<24].reduce(UInt32(0)) { $0 << 8 | UInt32($1) }
899 let ap = try KeyDerivation.deriveMasterKey(
900 password: pin, salt: salt,
901 parameters: KDFParameters(memoryKiB: memory, passes: passes, lanes: bytes[24])
902 )

906 do {
907 blob = try store.read(Account.rootByPIN + suffix, context: store.applicationPasswordContext(ap))
908 } catch StoreError.authenticationFailed {
909 throw Failure.wrongPIN(remainingBeforeWipe: nil)
910 }
911 guard let blob else { throw Failure.notSetUp }

913 if version == 2 {

921 } else {

1061/// O rk.and inteiro: o cabecalho e, para cada camada de fora da politica, o mesmo miolo
1062/// (`ECIES_{kand.pin}(0x01 ‖ ChaChaPoly(K_int, RK))`) embrulhado pela chave dela.
1063///
1064/// Formato 4: cada camada presente, na ordem `kand.bio`, `kand.qualquer`, como tamanho (2,
1065/// big-endian) ‖ ECIES. Camadas que nao batem com o corpo (sobra, falta, tamanho zero): o
1066/// registro nao e este. Formato 3: so o ECIES da `kand.bio`, sem tamanho.
1067struct HardenedRecord: Equatable {
1068 var header: HardenedHeader
1069 var layers: [WrapSlot: Data]
1070
1071 var encoded: Data {
1072 var out = Data(header.encoded)
1073 if header.version == HardenedHeader.legacyVersion {
1074 out.append(layers[.hardenedBiometry] ?? Data())
1075 return out
1076 }
1077 for slot in header.faceChange.layers {
1078 let blob = layers[slot] ?? Data()
1079 out.append(contentsOf: UInt16(clamping: blob.count).bigEndianByteArray)
1080 out.append(blob)
1081 }
1082 return out
1083 }

1085 static func parse(_ data: Data) throws -> HardenedRecord {
1086 let bytes = [UInt8](data)
1087 guard let header = HardenedHeader(bytes) else { throw RootKeyVault.Failure.storage }
1088 var rest = bytes[header.byteCount...]
1089 if header.version == HardenedHeader.legacyVersion {
1090 guard !rest.isEmpty else { throw RootKeyVault.Failure.storage }
1091 return HardenedRecord(header: header, layers: [.hardenedBiometry: Data(rest)])
1092 }
1093 var layers: [WrapSlot: Data] = [:]
1094 for slot in header.faceChange.layers {
1095 guard rest.count >= 2 else { throw RootKeyVault.Failure.storage }
1096 let size = Int(rest[rest.startIndex]) << 8 | Int(rest[rest.startIndex + 1])
1097 rest = rest.dropFirst(2)
1098 guard size > 0, rest.count >= size else { throw RootKeyVault.Failure.storage }
1099 layers[slot] = Data(rest.prefix(size))
1100 rest = rest.dropFirst(size)
1101 }
1102 guard rest.isEmpty else { throw RootKeyVault.Failure.storage }
1103 return HardenedRecord(header: header, layers: layers)
1104 }

1421 /// vigia na mesma sessao do rosto, sem segundo Face ID; um miolo diferente nela e um
1422 /// registro que nao e o gravado. Nao conta nem grava: quem chama decide. Privado: nada
1423 /// fora do cofre abre "o rosto ja passou".

1426 let face = header.faceChange.faceKey
1427 guard let outer = record.layers[face] else { throw Failure.storage }
1428 let middle: SecureBytes

1431 if header.faceChange == .wait, let watch = record.layers[.hardenedBiometry] {
1432 let pair = try wrapper.unwrap(outer, slot: face, reason: reason, watching: watch, with: .hardenedBiometry)
1433 middle = pair.0
1434 seen = pair.1
1435 } else {
1436 middle = try wrapper.unwrap(outer, slot: face, reason: reason)
1437 }
1438 } catch EnclaveError.failed {

1440 // conta. A chave que nao existe mais e o cofre perdido.
1441 if wrapper.probeKey(face) == .absent { throw Failure.hardenedSlotLost }
1442 throw Failure.hardenedFaceKeyFailed
1443 } catch {

1445 // diferente com a chave presente pode ser so deriva do sistema e conta como o
1446 // rosto que nao confirmou (revisao do modo reforcado, M6).
1447 if wrapper.probeKey(face) == .absent { throw Failure.hardenedSlotLost }
1448 throw Failure.biometryNotConfirmed(remainingBeforeWipe: nil)
1449 }

1451 switch seen {
1452 case .opened(let other)?:
1453 let equal = Hash.constantTimeEqual(other, middle)
1454 other.wipe()

1456 middle.wipe()
1457 throw Failure.storage
1458 }
1459 watch = .same
1460 case .gone?: watch = .gone
1461 case .unknown?: watch = .unknown
1462 case nil: watch = nil
1463 }
1464 do {

1466 } catch {

1496 /// O que a vigia do `.wait` viu, ja com a RK aberta (PIN e rosto certos).
1497 ///
1498 /// - Mesmo cadastro: a vigia abriu o mesmo miolo, ou nao respondeu e o hash e o gravado.
1499 /// O hash e regravado, e uma espera vencida termina (um realinhamento que caiu antes de
1500 /// apagar a espera).
1501 /// - Troca: a chave da vigia nao existe mais, ou nao respondeu e o hash e outro (ou falta):
1502 /// sem espera, ela comeca, conferida; com a espera correndo, nada muda (so abrir o app
1503 /// chega aqui); vencida para o cadastro que a comecou, a vigia realinha; vencida para
1504 /// outro, outra espera comeca. Fora de abrir o app, a espera comecada agora recusa.
1505 ///
1506 /// Devolve se o cadastro e o conhecido.
1507 private func settleWatchLocked(_ record: HardenedRecord, opened: HardenedOpening, hold: HardenedHold?, use: HardenedUse) throws -> Bool {
1508 let now = hardenedDomainState()
1509 let changed: Bool
1510 switch opened.watch {
1511 case .same?:
1512 changed = false
1513 case .unknown?:
1514 let saved = try? store.read(Account.hardenedState, context: nil)
1515 changed = saved == nil || now == nil || saved != now
1516 case .gone?, nil:
1517 changed = true
1518 }
1519 if !changed {
1520 // So a prova do SEP (a vigia abriu o mesmo miolo) regrava o hash e termina uma
1521 // espera vencida. O hash sozinho nunca encerra espera.
