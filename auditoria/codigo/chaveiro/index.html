29/// A protecao de cada item. So existem estas duas, e as duas sao
30/// `WhenPasscodeSetThisDeviceOnly`: o item some se o dono tirar o codigo do iPhone,
31/// nunca sincroniza, nunca vai para backup, e nunca abre em outro aparelho.
32///
33/// Proibidos no projeto inteiro, por verificar.sh: `.userPresence`, `.devicePasscode`,
34/// `.biometryAny`, `AfterFirstUnlock` e `Always`. Quem viu o codigo do iPhone passa
35/// por `.userPresence`; com `.biometryAny` o ladrao cadastra o proprio rosto.
36public enum ItemProtection: Sendable {
37 /// Item comum, legivel com o aparelho desbloqueado.
38 case standard
39 /// Item que o proprio chaveiro so devolve com a senha de aplicativo certa
40 /// (`kSecAccessControlApplicationPassword`). E o que prende o PIN ao aparelho:
41 /// nao existe verificador de PIN gravado em lugar nenhum; o chaveiro recusa.
42 case applicationPassword
43}

67 ]

70 public func read(_ account: String, context: LAContext?) throws -> Data? {
71 var query = baseQuery(account)
72 query[kSecReturnData as String] = kCFBooleanTrue
73 query[kSecMatchLimit as String] = kSecMatchLimitOne
74 if let context { query[kSecUseAuthenticationContext as String] = context }
75 var result: AnyObject?
76 let status = SecItemCopyMatching(query as CFDictionary, &result)
77 release(context)
78 switch status {
79 case errSecSuccess: return result as? Data
80 case errSecItemNotFound: return nil
81 case errSecAuthFailed, errSecUserCanceled, errSecInteractionNotAllowed: throw StoreError.authenticationFailed
82 default: throw StoreError.unexpected(status)
83 }
84 }
85
86 public func add(_ data: Data, account: String, protection: ItemProtection, context: LAContext?) throws {
87 var query = baseQuery(account)
88 query[kSecValueData as String] = data
89 switch protection {
90 case .standard:
91 query[kSecAttrAccessible as String] = Self.accessibility
92 case .applicationPassword:
93 var error: Unmanaged<CFError>?
94 guard let access = SecAccessControlCreateWithFlags(nil, Self.accessibility, [.applicationPassword], &error) else {
95 throw StoreError.unexpected(errSecParam)
96 }
97 query[kSecAttrAccessControl as String] = access
98 }
99 if let context { query[kSecUseAuthenticationContext as String] = context }
100 let status = SecItemAdd(query as CFDictionary, nil)
101 release(context)
102 switch status {
103 case errSecSuccess: return
104 case errSecDecode, errSecNotAvailable: throw StoreError.passcodeNotSet
105 default: throw StoreError.unexpected(status)
106 }
107 }

109 /// Altera no lugar. O contador de tentativas usa isto e nunca apagar e recriar:
110 /// um `SecItemAdd` que falha depois de um `SecItemDelete` (disco cheio) zerava o
111 /// atraso no Escalibur. Aqui, se a gravacao falha, a tentativa nao e avaliada.
112 public func update(_ data: Data, account: String) throws {
113 let attributes = [kSecValueData as String: data]
114 let status = SecItemUpdate(baseQuery(account) as CFDictionary, attributes as CFDictionary)
115 guard status == errSecSuccess else { throw StoreError.unexpected(status) }
116 }

135 public func probe(_ account: String) -> ItemState {
136 var query = baseQuery(account)
137 query[kSecReturnAttributes as String] = kCFBooleanTrue
138 let context = LAContext()
139 context.interactionNotAllowed = true
140 query[kSecUseAuthenticationContext as String] = context
141 let status = SecItemCopyMatching(query as CFDictionary, nil)
142 context.invalidate()
143 switch status {
144 case errSecSuccess, errSecInteractionNotAllowed, errSecAuthFailed: return .present
145 case errSecItemNotFound: return .absent
146 default: return .unknown
147 }
148 }

150 /// O contexto e usado numa chamada so e invalidado logo depois (ver `read` e
151 /// `add`), para a senha de aplicativo nao ficar retida num objeto vivo.
152 public func applicationPasswordContext(_ password: SecureBytes) throws -> LAContext? {
153 let context = LAContext()
154 context.interactionNotAllowed = true
155 let ok = password.withUnsafeData { context.setCredential($0, type: .applicationPassword) }
156 guard ok else {
157 #if targetEnvironment(simulator)
158 // O simulador recusa a credencial e grava o item sem senha de aplicativo.
159 // La, quem recusa o PIN errado e so a camada interna do RootKeyVault.
160 context.invalidate()
161 return nil
162 #else
163 context.invalidate()
164 throw StoreError.unexpected(errSecParam)
165 #endif
166 }
167 return context
168 }
169
170 private func release(_ context: LAContext?) {
171 guard let context else { return }
172 context.setCredential(nil, type: .applicationPassword)
173 context.invalidate()
174 }
