144 /// A regra de todo endereco: https, host da lista, a porta 443 (ou a do host, acima) e,
145 /// quando o host tem, so o caminho dele.
146 static func permits(_ url: URL, allowedHosts: Set<String>) -> Bool {
147 guard url.scheme == "https", let host = url.host, allowedHosts.contains(host) else { return false }
148 guard url.port == nil || url.port == 443 || url.port == extraPorts[host] else { return false }
149 return pathPrefixes[host].map { url.path.hasPrefix($0) } ?? true
150 }

152 /// O corpo e lido em fluxo e cortado ao passar do limite. Resposta sem
153 /// Content-Length (chunked) nao chega inteira na memoria antes de ser medida.
154 ///
155 /// Uma tarefa de dados com o delegado da sessao, e nao `bytes(for:)`: em algumas
156 /// versoes do macOS (a do runner do CI) o fluxo de `bytes(for:)` parava de entregar
157 /// o corpo, e o delegado classico mede cada pedaco igual em todas.
158 private func perform(_ request: URLRequest, limit: Int = HTTPClient.maxResponseBytes) async throws -> Data {
159 guard let url = request.url, Self.permits(url, allowedHosts: allowedHosts) else { throw Failure.hostNotAllowed }
160 guard !isQuiet else { throw Failure.offline }
161 let task = session.dataTask(with: request)
162 do {
163 return try await withTaskCancellationHandler {
164 try await withCheckedThrowingContinuation { continuation in
165 // A conferencia do silencio e o registro sob a mesma trava: uma tela de
166 // segredo que abre agora cancela esta tarefa, ou ela nem comeca.
167 let paused = lock.withLock { () -> Bool in
168 guard quietHolds == 0 else { return true }
169 transfers[ObjectIdentifier(task)] = Transfer(continuation, task: task, limit: limit)
170 return false
171 }
172 if paused {
173 continuation.resume(throwing: Failure.offline)
174 return
175 }
176 task.resume()
177 }
178 } onCancel: {
179 task.cancel()
180 }
181 } catch {
182 throw error
183 }
184 }

203 /// Quantas telas de segredo estao abertas agora (`beginQuiet`). Com alguma aberta,
204 /// nenhuma requisicao sai do aparelho, e as que estavam no ar sao canceladas: a tela

216 return transfers.values.map(\.task)

253 /// Status e tamanho anunciado sao conferidos antes de qualquer byte do corpo.
254 public func urlSession(
255 _ session: URLSession, dataTask: URLSessionDataTask, didReceive response: URLResponse,
256 completionHandler: @escaping @Sendable (URLSession.ResponseDisposition) -> Void
257 ) {
258 guard let transfer = transfer(dataTask) else { return completionHandler(.cancel) }
259 let refusal: Failure?
260 if let http = response as? HTTPURLResponse {
261 if (300..<400).contains(http.statusCode) { refusal = .redirectRefused }
262 else if !(200..<300).contains(http.statusCode) { refusal = .status(http.statusCode) }
263 else if http.expectedContentLength > Int64(transfer.limit) { refusal = .tooLarge }
264 else { refusal = nil }
265 } else {
266 refusal = .invalidResponse
267 }
268 lock.withLock {
269 transfer.failure = refusal
270 if refusal == nil, response.expectedContentLength > 0 {
271 transfer.data.reserveCapacity(Int(response.expectedContentLength))
272 }
273 }
274 completionHandler(refusal == nil ? .allow : .cancel)
275 }

277 public func urlSession(_ session: URLSession, dataTask: URLSessionDataTask, didReceive data: Data) {
278 guard let transfer = transfer(dataTask) else { return }
279 let over = lock.withLock { () -> Bool in
280 guard transfer.failure == nil else { return false }
281 transfer.data.append(data)
282 guard transfer.data.count > transfer.limit else { return false }
283 transfer.failure = .tooLarge
284 transfer.data = Data()
285 return true
286 }
287 if over { dataTask.cancel() }

305/// Um conjunto de provedores equivalentes para o mesmo servico, com contingencia.
306///
307/// Circuit breaker simples: tres falhas seguidas tiram o provedor por 60 segundos.
308/// A ordem de preferencia e fixa no codigo; nenhuma resposta de rede acrescenta
309/// provedor.
310public actor ProviderPool {
311 public struct Provider: Sendable, Hashable {
312 public let name: String
313 public let baseURL: URL
314 /// A empresa que opera o no. Dois enderecos do mesmo dono (os dois servidores da
315 /// Ripple) contam como um so no consenso: um dono que mente, mente nos dois.

329 public let providers: [Provider]
330 private var failures: [String: Int] = [:]

333 public init(_ providers: [Provider]) {
334 self.providers = Self.spreadingOperators(providers)

337 /// Os provedores disponiveis agora, na ordem de preferencia, com donos diferentes na
338 /// frente: quem pega os dois primeiros para um consenso pega dois donos.
339 public func available(now: Date = .now) -> [Provider] {
340 let live = providers.filter { (benchedUntil[$0.name] ?? .distantPast) <= now }
341 return Self.spreadingOperators(live.isEmpty ? providers : live)
342 }
343
344 /// A mesma ordem, mas cada dono aparece uma vez antes de qualquer repeticao.
345 static func spreadingOperators(_ providers: [Provider]) -> [Provider] {
346 var seen = Set<String>()
347 var first: [Provider] = []
348 var repeated: [Provider] = []
349 for provider in providers {
350 if seen.insert(provider.operatorName).inserted { first.append(provider) } else { repeated.append(provider) }
351 }
352 return first + repeated
353 }
354
355 public func reportSuccess(_ provider: Provider) {
356 failures[provider.name] = 0
357 }
358
359 public func reportFailure(_ provider: Provider, now: Date = .now) {
360 let count = (failures[provider.name] ?? 0) + 1
361 failures[provider.name] = count
362 if count >= 3 {
363 benchedUntil[provider.name] = now.addingTimeInterval(60)
364 failures[provider.name] = 0
365 }
366 }
367
368 /// A falha com o motivo. Recusa por limite vai direto para o banco, por mais tempo:
369 /// insistir num provedor que ja disse "chega" so prolonga o bloqueio do IP (a
370 /// Blockchair passa do 429 ao 430, "IP temporary blacklisted"). O resto segue a regra
371 /// das tres falhas seguidas.
372 public func reportFailure(_ provider: Provider, error: Error, now: Date = .now) {
373 guard let pause = Self.benchTime(for: error) else { return reportFailure(provider, now: now) }
374 let until = now.addingTimeInterval(pause)
375 benchedUntil[provider.name] = max(benchedUntil[provider.name] ?? until, until)
376 failures[provider.name] = 0
377 }
378
379 /// Quanto tempo cada recusa deixa o provedor de fora. 429 e o limite por IP; 430
380 /// (Blockchair) e 403 (Cloudflare) sao bloqueio do IP, mais demorado; um provedor
381 /// fora do ar custa o timeout inteiro a cada pergunta.
382 public static func benchTime(for error: Error) -> TimeInterval? {
383 switch error as? HTTPClient.Failure {
384 case .status(429): return 5 * 60
385 case .status(430), .status(403): return 15 * 60
386 case .timeout: return 2 * 60
387 default: return nil
388 }
389 }

391 /// So para os testes e para a tela: ate quando o provedor esta no banco.
392 public func benched(_ provider: Provider) -> Date? { benchedUntil[provider.name] }

394 /// Tenta cada provedor em ordem ate um responder.
395 public func first<T: Sendable>(_ operation: @Sendable (Provider) async throws -> T) async throws -> T {
396 var lastError: Error = HTTPClient.Failure.offline
397 for provider in available() {
398 do {
399 let value = try await operation(provider)
400 reportSuccess(provider)
401 return value
402 } catch {
403 reportFailure(provider)
404 lastError = error
405 }
406 }
407 throw lastError
408 }

410 /// Pergunta a provedores distintos e exige que dois concordem. Nonce, sequence e
411 /// saldo que vao para uma transacao passam por aqui (docs/seguranca.md §5.5).
412 ///
413 /// Nunca aceita uma resposta so: com menos de dois provedores respondendo, falha.
414 /// Os que estao no banco do circuit breaker entram no fim da fila, porque um
415 /// consenso que so tem um provedor vivo nao e consenso. Dois enderecos do mesmo
416 /// dono valem uma resposta so.
417 public func agreeing<T: Sendable & Equatable>(_ operation: @Sendable (Provider) async throws -> T) async throws -> T {
418 let live = available()
419 let candidates = Self.spreadingOperators(live + providers.filter { !live.contains($0) })
420 var answers: [(operatorName: String, value: T)] = []
421 for provider in candidates where !answers.contains(where: { $0.operatorName == provider.operatorName }) {
422 do {
423 let value = try await operation(provider)
424 reportSuccess(provider)
425 if answers.contains(where: { $0.value == value }) { return value }
426 answers.append((provider.operatorName, value))
427 } catch {
428 reportFailure(provider)
429 }
430 }
431 throw ConsensusFailure(answers: answers.count)
432 }
433}

451 public static func call<T: Decodable>(_ url: URL, method: String, params: [JSONValue], as type: T.Type, client: HTTPClient = .shared) async throws -> T {
452 let body: [String: JSONValue] = [
453 "jsonrpc": .string("2.0"), "id": .number(1), "method": .string(method), "params": .array(params),
454 ]
455 let data = try await client.post(url, json: try JSONEncoder().encode(body))
456 let envelope: Envelope<T>
457 do { envelope = try JSONDecoder().decode(Envelope<T>.self, from: data) } catch { throw HTTPClient.Failure.decoding(method) }
458 if let error = envelope.error { throw error }
459 guard let result = envelope.result else { throw HTTPClient.Failure.invalidResponse }
460 return result
461 }

464/// Um valor JSON qualquer, para montar parametros sem `[String: Any]`.
465public enum JSONValue: Codable, Sendable, Equatable {
466 case string(String)
467 case number(Double)
468 case bool(Bool)
469 case array([JSONValue])
470 case object([String: JSONValue])
471 case null
472
473 public func encode(to encoder: Encoder) throws {
474 var container = encoder.singleValueContainer()
475 switch self {
476 case .string(let value): try container.encode(value)
477 case .number(let value):
478 if value == value.rounded(), abs(value) < 1e15 { try container.encode(Int64(value)) } else { try container.encode(value) }
479 case .bool(let value): try container.encode(value)
480 case .array(let value): try container.encode(value)
481 case .object(let value): try container.encode(value)
482 case .null: try container.encodeNil()
483 }
484 }
485
486 public init(from decoder: Decoder) throws {
487 let container = try decoder.singleValueContainer()
488 if container.decodeNil() { self = .null }
489 else if let value = try? container.decode(Bool.self) { self = .bool(value) }
490 else if let value = try? container.decode(Double.self) { self = .number(value) }
491 else if let value = try? container.decode(String.self) { self = .string(value) }
492 else if let value = try? container.decode([JSONValue].self) { self = .array(value) }
493 else { self = .object(try container.decode([String: JSONValue].self)) }
494 }
495
496 public var stringValue: String? { if case .string(let v) = self { return v }; return nil }
497 public var doubleValue: Double? { if case .number(let v) = self { return v }; return nil }
498 public var boolValue: Bool? { if case .bool(let v) = self { return v }; return nil }
499 public subscript(key: String) -> JSONValue? { if case .object(let o) = self { return o[key] }; return nil }
500 public subscript(index: Int) -> JSONValue? {
501 if case .array(let a) = self, a.indices.contains(index) { return a[index] }
502 return nil
503 }
504 public var arrayValue: [JSONValue]? { if case .array(let a) = self { return a }; return nil }
505}
