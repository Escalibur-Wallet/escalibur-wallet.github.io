8/// O formato do arquivo `.esclbr`, campo a campo.
9///
10/// Duas exigencias se cruzam aqui, e a segunda molda a primeira:
11///
12/// - Do desenho criptografico: envelope explicito. A chave que cifra a frase (`DEK`)
13/// nasce do gerador do sistema e nunca da senha; ela e guardada embrulhada por uma
14/// chave derivada da senha (`KEK`). Trocar a senha re-embrulha 32 bytes em vez de
15/// re-cifrar o cofre inteiro.
16///
17/// - Do desenho de custodia: negacao plausivel de verdade. Um cofre isca guardado
18/// como "mais um arquivo" nao nega nada, porque a existencia do arquivo ja e a
19/// prova. Entao **todo arquivo tem sempre quatro compartimentos do mesmo tamanho**,
20/// e os compartimentos sem uso ficam cheios de bytes aleatorios. Cifra autenticada
21/// e ruido sao indistinguiveis, entao quem obriga o dono a abrir um compartimento
22/// nao consegue provar que existe outro.
23///
24/// Consequencias que valem por si:
25///
26/// - **Todo cofre tem exatamente 16.504 bytes.** O tamanho do arquivo nao conta
27/// quantas palavras a frase tem, se ha anotacao, nem quantos compartimentos estao
28/// em uso. Metadado de tamanho e de graca para o atacante escolher qual arquivo
29/// atacar primeiro, e aqui ele nao existe.
30/// - **O cabecalho em claro nao tem nada com significado.** Sem nome, sem data, sem
31/// dica de senha, sem contador. So o que a mecanica de abertura exige. O nome do
32/// cofre mora dentro do compartimento cifrado, porque "Ledger principal" escrito em
33/// claro ja diz a quem pega o arquivo que vale a pena insistir.
34///
35/// O formato e publico de proposito. Um cofre que so este app abre e um cofre com
36/// prazo de validade, e a especificacao em docs/formato.md mais o decifrador de
37/// referencia existem para que o dono nunca dependa da loja de aplicativos.

116 func encoded() -> [UInt8] {
117 var bytes = [UInt8](repeating: 0, count: VaultFormat.headerLength)
118 bytes.replaceSubrange(Header.magic..<(Header.magic + 6), with: VaultFormat.magic)
119 bytes[Header.formatVersion] = VaultFormat.formatVersion
120 bytes[Header.suiteID] = VaultFormat.suiteID
121
122 let uuid = withUnsafeBytes(of: vaultID.uuid) { Array($0) }
123 bytes.replaceSubrange(Header.vaultUUID..<(Header.vaultUUID + 16), with: uuid)
124
125 bytes[Header.kdfID] = VaultFormat.kdfID
126 bytes.replaceSubrange(
127 Header.kdfMemoryKiB..<(Header.kdfMemoryKiB + 4),
128 with: kdf.memoryKiB.bigEndianBytes
129 )
130 bytes.replaceSubrange(
131 Header.kdfPasses..<(Header.kdfPasses + 4),
132 with: kdf.passes.bigEndianBytes
133 )
134 bytes[Header.kdfLanes] = kdf.lanes
135 bytes.replaceSubrange(
136 Header.kdfSalt..<(Header.kdfSalt + VaultFormat.saltLength),
137 with: kdfSalt
138 )
139 bytes[Header.bindingID] = binding.rawValue
140 bytes.replaceSubrange(
141 Header.bindingPublicKey..<(Header.bindingPublicKey + 65),
142 with: bindingPublicKey
143 )
144 bytes[Header.cloudFlag] = cloudEnabled ? 0x01 : 0x00
145 return bytes
146 }

148 static func decode(_ bytes: [UInt8]) throws -> Header_ {
149 guard bytes.count >= VaultFormat.headerLength else {
150 throw CryptoError.malformedVault("arquivo curto demais para ser um cofre")
151 }
152 guard Array(bytes[Header.magic..<(Header.magic + 6)]) == VaultFormat.magic else {
153 throw CryptoError.malformedVault("este arquivo não é um cofre do Escalibur")
154 }
155 // Versao exata, e nao "ate a atual": aceitar 0x00 e trata-lo como v1
156 // daria significado retroativo a um valor que a especificacao reservou
157 // como invalido. Nao e vetor de downgrade, porque o cabecalho inteiro
158 // entra no AAD, mas um arquivo escrito a mao com 0x00 nao pode passar.
159 guard bytes[Header.formatVersion] == VaultFormat.formatVersion else {
160 throw CryptoError.malformedVault(
161 bytes[Header.formatVersion] > VaultFormat.formatVersion
162 ? "cofre de uma versão mais nova do formato; atualize o app"
163 : "versão de formato inválida"
164 )
165 }
166 guard bytes[Header.suiteID] == VaultFormat.suiteID,
167 bytes[Header.kdfID] == VaultFormat.kdfID
168 else {
169 throw CryptoError.malformedVault("conjunto de algoritmos desconhecido")
170 }
171
172 let uuidBytes = Array(bytes[Header.vaultUUID..<(Header.vaultUUID + 16)])
173 let vaultID = uuidBytes.withUnsafeBytes { raw in
174 UUID(uuid: raw.load(as: uuid_t.self))
175 }
176
177 let kdf = KDFParameters(
178 memoryKiB: UInt32(
179 bigEndian: Array(bytes[Header.kdfMemoryKiB..<(Header.kdfMemoryKiB + 4)])
180 ),
181 passes: UInt32(
182 bigEndian: Array(bytes[Header.kdfPasses..<(Header.kdfPasses + 4)])
183 ),
184 lanes: bytes[Header.kdfLanes]
185 )
186 // A faixa e conferida aqui, antes de qualquer alocacao. Um arquivo com
187 // `m = 4 GiB` escrito a mao mataria o app por falta de memoria, e isso
188 // seria negacao de servico de graca.
189 guard kdf.isWithinAcceptedRange else {
190 throw CryptoError.malformedVault("parâmetros de derivação fora da faixa aceita")
191 }
192
193 guard let binding = Binding(rawValue: bytes[Header.bindingID]) else {
194 throw CryptoError.malformedVault("tipo de vínculo desconhecido")
195 }
196
197 let cloudEnabled = bytes[Header.cloudFlag] == 0x01
198 // Um cofre preso a este aparelho e ao mesmo tempo marcado para a nuvem
199 // seria lixo irrecuperavel do outro lado. So um adversario produz essa
200 // combinacao, entao ela e recusada.
201 guard !(binding == .secureEnclave && cloudEnabled) else {
202 throw CryptoError.malformedVault("cofre com vínculo de aparelho não vai para a nuvem")
203 }
204
205 return Header_(
206 vaultID: vaultID,
207 kdf: kdf,
208 kdfSalt: Array(bytes[Header.kdfSalt..<(Header.kdfSalt + VaultFormat.saltLength)]),
209 binding: binding,
210 bindingPublicKey: Array(
211 bytes[Header.bindingPublicKey..<(Header.bindingPublicKey + 65)]
212 ),
213 cloudEnabled: cloudEnabled
214 )
215 }

218 // MARK: Material de chave

220 /// `KEK` do compartimento `index`.
221 ///
222 /// O salt do Argon2 e do arquivo, nao do compartimento, e isso e deliberado: com
223 /// um salt por compartimento, abrir um cofre exigiria rodar o Argon2 quatro
224 /// vezes, o que a 1,5 s cada tornaria a abertura insuportavel. Com um salt so, a
225 /// derivacao cara acontece uma vez e os quatro compartimentos sao testados com
226 /// HKDF, que e barato.
227 public static func encryptionKey(
228 masterKey: SecureBytes,
229 bindingSecret: SecureBytes?,
230 slotSalt: [UInt8],
231 slotIndex: Int,
232 vaultID: UUID
233 ) -> SymmetricKey {
234 var ikm = [UInt8]()
235 defer { ikm.resetBytes() }
236
237 masterKey.withUnsafeBytes { ikm.append(contentsOf: $0) }
238 // O segredo do aparelho entra na **mesma** chave, e nao num segundo
239 // compartimento. Num segundo compartimento, o compartimento da senha
240 // continuaria atacavel offline e o vinculo nao valeria nada.
241 bindingSecret?.withUnsafeBytes { ikm.append(contentsOf: $0) }
242
243 var info = Array("escalibur/v1/kek".utf8)
244 info.append(UInt8(slotIndex))
245 info.append(contentsOf: withUnsafeBytes(of: vaultID.uuid) { Array($0) })
246
247 return HKDF<SHA256>.deriveKey(
248 inputKeyMaterial: SymmetricKey(data: ikm),
249 salt: slotSalt,
250 info: info,
251 outputByteCount: keyLength
252 )
253 }
254
255 /// Subchave de uma operacao de cifra.
256 ///
257 /// O nonce e sempre zero, e isso e seguro porque a **chave** nunca se repete: ela
258 /// nasce de um salt de 32 bytes aleatorios gerado a cada gravacao. E o padrao do
259 /// `secretstream` do libsodium, e resolve de forma definitiva o problema que
260 /// derrubaria um contador de nonce: restaurar um backup antigo do iPhone faz o
261 /// contador voltar no tempo e reemitir um nonce ja usado, o que em cifra
262 /// autenticada nao vaza so a mensagem, vaza a chave de autenticacao.
263 public static func messageKey(from key: SymmetricKey, salt: [UInt8], purpose: String) -> SymmetricKey {
264 HKDF<SHA256>.deriveKey(
265 inputKeyMaterial: key,
266 salt: salt,
267 info: Array(purpose.utf8),
268 outputByteCount: keyLength
269 )
270 }
271
272 /// Doze zeros, sempre. E seguro porque a **chave** nunca se repete: cada
273 /// operacao deriva a sua a partir de um salt sorteado na hora.
274 public static let zeroNonce = try! ChaChaPoly.Nonce(
275 data: Data(repeating: 0, count: 12)
276 )

281extension UInt32 {
282 public var bigEndianBytes: [UInt8] {
283 [UInt8(truncatingIfNeeded: self >> 24),

285 UInt8(truncatingIfNeeded: self >> 8),

289 public init(bigEndian bytes: [UInt8]) {

291 self = (UInt32(bytes[0]) << 24) | (UInt32(bytes[1]) << 16)
292 | (UInt32(bytes[2]) << 8) | UInt32(bytes[3])
293 }
