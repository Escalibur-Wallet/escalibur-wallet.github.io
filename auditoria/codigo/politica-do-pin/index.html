10/// As regras do PIN: a forma, a escada de atraso e o relogio que sobrevive a
11/// reinicio sem trancar o dono por dias.
12public enum PINPolicy {
13 /// O tamanho padrao, e o de todo PIN gravado antes de existir a escolha.
14 public static let digits = 6
15
16 /// Os tamanhos que o dono pode escolher (pedido do dono, 16/10/2026: 4 tambem). Com 4,
17 /// sao 10^4 combinacoes em vez de 10^6: contra quem tem execucao de codigo no
18 /// aparelho, minutos em vez de horas (docs/seguranca.md §2.8).
19 public static let lengths = [4, 6]
20
21 /// O PIN e do dono: qualquer combinacao de 4 ou 6 digitos vale, inclusive 1111 e
22 /// 123456. Decisao do dono em 2026-09-26, que retirou a lista de PINs faceis. So a
23 /// forma e conferida: 4 ou 6 digitos de 0 a 9. Contra quem tenta pela interface
24 /// ficam a escada de atraso e, se o dono ligar, o apagar apos erros.
25 public static func isWellFormed(_ digits: [UInt8]) -> Bool {
26 lengths.contains(digits.count) && digits.allSatisfy { $0 <= 9 }
27 }
28
29 /// A escada de atraso, em segundos, depois de `failures` erros seguidos.
30 ///
31 /// Do nono erro em diante a espera dobra a cada erro, de 3 horas ate 7 dias (auditoria
32 /// 4, C7-04). Com o teto antigo de 3 horas, quem tem o iPhone tentava cerca de 2.900
33 /// PINs por ano; com o dobro, 18 em 30 dias e 66 em um ano.
34 public static func delay(afterFailures failures: UInt32) -> TimeInterval {
35 switch failures {
36 case 0...2: return 0
37 case 3: return 5
38 case 4: return 15
39 case 5: return 60
40 case 6: return 5 * 60
41 case 7: return 15 * 60
42 case 8: return 60 * 60
43 default:
44 let doubled = 3 * 60 * 60 * pow(2, Double(min(failures - 9, 16)))
45 return min(doubled, maxDelay)
46 }
47 }
48
49 /// O teto da escada: 7 dias por tentativa.
50 public static let maxDelay: TimeInterval = 7 * 24 * 60 * 60
51
52 /// A espera do A1 depois que o Face ID cai sem o dono pedir (o cadastro de rostos mudou,
53 /// o Face ID foi tirado do app ou do iPhone): durante ela, ver as palavras, lacrar o
54 /// envelope e remover carteira ficam travados. Quem viu o codigo do iPhone e o PIN
55 /// cadastra o proprio rosto e derruba o slot; sem a espera, isso trocava "PIN e rosto"
56 /// por "so o PIN" na hora (auditoria 4, A1).
57 public static let biometryLossHold: TimeInterval = 24 * 60 * 60
58
59 /// Quantos erros podem apagar tudo, a escolha do dono (ou nunca).
60 public static let wipeOptions: [UInt32] = [5, 10, 15, 20]
61
62 // MARK: Relogios
63
64 /// Tempo desde o boot, contando o tempo dormindo, e que nao anda para tras quando
65 /// alguem muda o relogio. O relogio de parede nunca entra na conta da espera:
66 /// adiantar nao encurta, atrasar nao tranca o dono por um ano.
67 public static var uptime: TimeInterval {
68 var time = timespec()
69 #if os(Android)
70 // No Linux o relogio que conta o tempo dormindo e o CLOCK_BOOTTIME.
71 clock_gettime(CLOCK_BOOTTIME, &time)
72 #else
73 clock_gettime(CLOCK_MONOTONIC_RAW, &time)
74 #endif
75 return TimeInterval(time.tv_sec) + TimeInterval(time.tv_nsec) / 1e9
76 }
77
78 /// Identidade deste boot, 16 bytes. Se mudou, o relogio monotonico recomecou do zero
79 /// e o prazo gravado com ele deixou de valer.
80 ///
81 /// No Mac e no simulador e o `kern.bootsessionuuid`, comparado byte a byte. No iPhone
82 /// a sandbox recusa essa leitura (EPERM, medido num iPhone 13 Pro Max com iOS 26.3):
83 /// vale o instante do boot (`kern.boottime`), marcado com `boottimeMark` e comparado
84 /// com folga de `boottimeTolerance` em `isSameBoot`, porque ele oscila um pouco com o
85 /// ajuste de hora pela rede.
86 public static var bootSession: [UInt8] {
87 #if os(Android)
88 // No Android, a identidade do boot que o sistema da ao app (KeyPlatform).
89 guard let boot = KeyPlatform.current.bootIdentity(), boot.count == 16, boot != unknownBoot else { return unknownBoot }
90 return boot
91 #else
92 if let uuid = bootSessionUUID { return withUnsafeBytes(of: uuid.uuid) { Array($0) } }
93 var time = timeval()
94 var size = MemoryLayout<timeval>.size
95 guard sysctlbyname("kern.boottime", &time, &size, nil, 0) == 0, time.tv_sec > 0 else { return unknownBoot }
96 let micros = UInt64(time.tv_sec) * 1_000_000 + UInt64(time.tv_usec)
97 return boottimeMark + (0..<8).map { UInt8(truncatingIfNeeded: micros >> (56 - 8 * $0)) }
98 #endif
99 }
100
101 #if !os(Android)
102 private static var bootSessionUUID: UUID? {
103 var size = 0
104 guard sysctlbyname("kern.bootsessionuuid", nil, &size, nil, 0) == 0, size > 0 else { return nil }
105 var buffer = [CChar](repeating: 0, count: size)
106 guard sysctlbyname("kern.bootsessionuuid", &buffer, &size, nil, 0) == 0 else { return nil }
107 return UUID(uuidString: String(decoding: buffer.prefix { $0 != 0 }.map { UInt8(bitPattern: $0) }, as: UTF8.self))
108 }
109 #endif
110
111 /// Os 8 primeiros bytes de uma identidade tirada do `kern.boottime` ("ESCBOOT1").
112 static let boottimeMark: [UInt8] = Array("ESCBOOT1".utf8)
113
114 /// Reiniciar move o instante do boot pelo tempo que o iPhone ficou ligado mais o da
115 /// propria reinicializacao, sempre bem mais que isto. O ajuste de hora pela rede
116 /// move milissegundos. Mudar a hora a mao parece um boot novo: o PIN volta a ser
117 /// pedido e a espera recomeca cheia, nunca o contrario.
118 static let boottimeTolerance: UInt64 = 10_000_000
119
120 /// As duas identidades sao do mesmo boot? Desconhecida nunca e.
121 public static func isSameBoot(_ a: [UInt8], _ b: [UInt8]) -> Bool {
122 guard a.count == 16, b.count == 16, a != unknownBoot, b != unknownBoot else { return false }
123 guard Array(a.prefix(8)) == boottimeMark, Array(b.prefix(8)) == boottimeMark else { return a == b }
124 let x = a.suffix(8).reduce(UInt64(0)) { $0 << 8 | UInt64($1) }
125 let y = b.suffix(8).reduce(UInt64(0)) { $0 << 8 | UInt64($1) }
126 return (x > y ? x - y : y - x) <= boottimeTolerance
127 }
128
129 /// Sem identidade de boot, cada leitura parece um boot novo: a espera recomeca
130 /// cheia. Falha para o lado do dono esperar mais, nunca menos.
131 public static let unknownBoot = [UInt8](repeating: 0, count: 16)
132
133 /// O registro de "PIN digitado neste boot": a identidade do boot e o tempo desde o
134 /// boot naquele instante (24 bytes).
135 ///
136 /// So o instante do boot (`kern.boottime`) deixava uma brecha: depois de reiniciar,
137 /// quem acertasse a hora a mao para o boot antigo cair dentro da folga religava o Face
138 /// ID sem o PIN (auditoria 3, B2). O tempo desde o boot recomeca do zero ao reiniciar
139 /// e nao anda com o relogio: o registro so vale se o de agora for maior.
140 public static func pinRecord(boot: [UInt8], uptime: TimeInterval) -> [UInt8] {
141 boot + withUnsafeBytes(of: uptime.bitPattern.bigEndian) { Array($0) }
142 }
143
144 /// Registro antigo, de so 16 bytes, nao vale: o PIN e pedido uma vez.
145 public static func pinRecordValid(_ saved: [UInt8], boot: [UInt8], uptime: TimeInterval) -> Bool {
146 guard saved.count == 24 else { return false }
147 let recordedUptime = TimeInterval(bitPattern: saved.suffix(8).reduce(UInt64(0)) { $0 << 8 | UInt64($1) })
148 guard recordedUptime.isFinite, recordedUptime >= 0 else { return false }
149 return isSameBoot(Array(saved.prefix(16)), boot) && uptime >= recordedUptime
150 }
151}

153/// O registro de tentativas, gravado no chaveiro.
154///
155/// Formato 2 (29 bytes): versao, erros (u32), prazo no relogio monotonico (f64) e a
156/// identidade do boot em que o prazo foi gravado (16 bytes).
157struct AttemptRecord: Equatable {
158 var failures: UInt32
159 var uptimeDeadline: TimeInterval
160 var bootSession: [UInt8]
161
162 static let zero = AttemptRecord(failures: 0, uptimeDeadline: 0, bootSession: PINPolicy.unknownBoot)
163 private static let version: UInt8 = 2
164
165 var encoded: Data {
166 var out = Data([Self.version])
167 out.append(contentsOf: failures.bigEndianByteArray)
168 out.append(contentsOf: uptimeDeadline.bitPattern.bigEndianByteArray)
169 out.append(contentsOf: bootSession)
170 return out
171 }
172
173 init(failures: UInt32, uptimeDeadline: TimeInterval, bootSession: [UInt8]) {
174 self.failures = failures
175 self.uptimeDeadline = uptimeDeadline
176 self.bootSession = bootSession
177 }
178
179 init?(_ data: Data) {
180 let bytes = [UInt8](data)
181 switch bytes.count {
182 case 29 where bytes[0] == Self.version:
183 failures = bytes[1..<5].reduce(0) { $0 << 8 | UInt32($1) }
184 uptimeDeadline = Double(bitPattern: bytes[5..<13].reduce(0) { $0 << 8 | UInt64($1) })
185 bootSession = Array(bytes[13..<29])
186 case 28:
187 // Formato 1, de antes da auditoria: so o numero de erros ainda vale. O
188 // boot desconhecido faz a espera recomecar cheia no proximo acesso.
189 failures = bytes[0..<4].reduce(0) { $0 << 8 | UInt32($1) }
190 uptimeDeadline = 0
191 bootSession = PINPolicy.unknownBoot
192 default:
193 return nil
194 }
195 }
196
197 /// Registro do erro numero `failures`, com o prazo contado a partir de agora.
198 static func after(failures: UInt32) -> AttemptRecord {
199 AttemptRecord(
200 failures: failures,
201 uptimeDeadline: PINPolicy.uptime + PINPolicy.delay(afterFailures: failures),
202 bootSession: PINPolicy.bootSession
203 )
204 }
205
206 /// O prazo foi gravado em outro boot (ou num formato antigo)?
207 var isFromAnotherBoot: Bool {
208 failures > 0 && !PINPolicy.isSameBoot(bootSession, PINPolicy.bootSession)
209 }
210
211 /// Quanto falta de espera. De outro boot, a resposta e a espera cheia ate alguem
212 /// regravar o prazo (`RootKeyVault.rebaseAttempts`); ler nunca escreve.
213 func remaining() -> TimeInterval {
214 let full = PINPolicy.delay(afterFailures: failures)
215 guard !isFromAnotherBoot else { return full }
216 return min(full, max(0, uptimeDeadline - PINPolicy.uptime))
217 }
218}

220/// O prazo da espera depois que o Face ID caiu (`PINPolicy.biometryLossHold`), gravado no
221/// chaveiro. 25 bytes: versao, prazo no relogio monotonico (f64) e a identidade do boot.
222/// Mesmo relogio do contador de tentativas: mudar a hora nao encurta, e depois de
223/// reiniciar a espera recomeca cheia.
224struct HoldRecord: Equatable {
225 var uptimeDeadline: TimeInterval
226 var bootSession: [UInt8]
227
228 private static let version: UInt8 = 1
229
230 static func startingNow() -> HoldRecord {
231 HoldRecord(uptimeDeadline: PINPolicy.uptime + PINPolicy.biometryLossHold, bootSession: PINPolicy.bootSession)
232 }
233
234 var encoded: Data {
235 var out = Data([Self.version])
236 out.append(contentsOf: uptimeDeadline.bitPattern.bigEndianByteArray)
237 out.append(contentsOf: bootSession)
238 return out
239 }
240
241 init(uptimeDeadline: TimeInterval, bootSession: [UInt8]) {
242 self.uptimeDeadline = uptimeDeadline
243 self.bootSession = bootSession
244 }
245
246 init?(_ data: Data) {
247 let bytes = [UInt8](data)
248 guard bytes.count == 25, bytes[0] == Self.version else { return nil }
249 uptimeDeadline = Double(bitPattern: bytes[1..<9].reduce(0) { $0 << 8 | UInt64($1) })
250 bootSession = Array(bytes[9..<25])
251 }
252
253 var isFromAnotherBoot: Bool { !PINPolicy.isSameBoot(bootSession, PINPolicy.bootSession) }
254
255 func remaining() -> TimeInterval {
256 let full = PINPolicy.biometryLossHold
257 guard !isFromAnotherBoot, uptimeDeadline.isFinite else { return full }
258 return min(full, max(0, uptimeDeadline - PINPolicy.uptime))
259 }
