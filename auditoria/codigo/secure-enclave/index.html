15/// Chaves de hardware que embrulham a chave raiz (RK).
16///
17/// Chaves P-256 no Secure Enclave:
18/// - `K_pin` (e `K_pin` da troca de PIN em andamento): `.privateKeyUsage` +
19/// `.applicationPassword`, com a senha derivada do PIN. O SE so desembrulha com ela:
20/// cada chute de PIN passa por este aparelho, um de cada vez (auditoria 3, A1;
21/// conferido num iPhone 13 Pro Max em 10/10/2026: a senha errada e recusada pelo
22/// proprio SE, 22 ms por tentativa).
23/// - `K_dev`: so `.privateKeyUsage`. A chave do PIN das versoes anteriores; e trocada
24/// pela `K_pin` no primeiro PIN certo e apagada.
25/// - `K_bio`: `.privateKeyUsage` + `.biometryCurrentSet`. Embrulha a RK para o
26/// caminho do Face ID. A decisao de liberar e do SEP, nao de um booleano no app, e
27/// cadastrar um rosto novo destroi a chave em vez de abri-la.
28/// - Modo reforcado: `kand.bio` (`.biometryCurrentSet`), `kand.qualquer`
29/// (`.biometryAny`, a unica chave com essa flag, que nunca embrulha a RK, so o miolo
30/// que ainda pede o PIN) e `kand.pin.a`/`kand.pin.b` (`.applicationPassword`). As
31/// flags de cada uma saem de `SecureEnclaveWrapper.flags(for:)`.
32///
33/// Nenhum embrulho abre fora deste aparelho: a parte privada nunca sai do SE.
34public protocol KeyWrapper: Sendable {
35 func createKey(_ slot: WrapSlot) throws
36 func hasKey(_ slot: WrapSlot) -> Bool
37 func deleteKey(_ slot: WrapSlot)
38 /// Embrulha com a chave de um slot sem senha (aparelho e rosto). Nos slots do PIN, use o
39 /// de baixo: buscar a chave sem a senha fazia o iOS pedir a senha na tela.
40 func wrap(_ secret: SecureBytes, slot: WrapSlot) throws -> Data
41 /// Embrulha com a chave de um slot do PIN, buscada com a senha de aplicativo e sem
42 /// interface nenhuma (visto no iPhone, 18/10/2026: ao ligar e desligar o Super Security,
43 /// o iOS abria "Digite a senha para Escalibur").
44 func wrap(_ secret: SecureBytes, slot: WrapSlot, password: Data) throws -> Data
45 /// Abre um embrulho. No slot biometrico, o sistema mostra o Face ID com
46 /// `reason`; a chamada bloqueia a thread ate a decisao, entao nunca no MainActor.
47 func unwrap(_ blob: Data, slot: WrapSlot, reason: String?) throws -> SecureBytes
48 /// Cria a chave do slot do PIN com senha de aplicativo. Substitui a que houver no
49 /// mesmo slot: o slot do PIN so e gravado quando nao existe.
50 func createKey(_ slot: WrapSlot, password: Data) throws
51 /// Desembrulha com a senha de aplicativo. Senha errada: `EnclaveError.wrongPassword`.
52 func unwrap(_ blob: Data, slot: WrapSlot, password: Data) throws -> SecureBytes
53 /// Existe, nao existe, ou o SE e o chaveiro nao conseguiram dizer. "Nao sei" nunca vira
54 /// "nao existe": e o que separa "o Face ID mudou" de "o iPhone esta bloqueado".
55 func probeKey(_ slot: WrapSlot) -> ItemState
56 /// Abre `blob` pelo rosto, com um Face ID so, e na mesma sessao, sem interface nenhuma,
57 /// tenta abrir `watch` com a chave `watchSlot` (a vigia do modo reforcado). A recusa do
58 /// rosto lanca como `unwrap(_:slot:reason:)`; a vigia nunca lanca: diz o que viu.
59 func unwrap(_ blob: Data, slot: WrapSlot, reason: String, watching watch: Data, with watchSlot: WrapSlot) throws -> (SecureBytes, WatchResult)
60}

73public enum WrapSlot: String, Sendable, CaseIterable {
74 case device = "kdev"
75 case biometry = "kbio"
76 /// A chave do slot principal do PIN, com senha de aplicativo.
77 case pin = "kpin"
78 /// A do slot da troca de PIN em andamento.
79 case pinPending = "kpin.novo"
80 /// Modo reforcado: a camada de fora do rk.and que acompanha o cadastro de rostos,
81 /// `.biometryCurrentSet`. A unica de fora em "Bloquear as carteiras"; a vigia em
82 /// "Esperar 24 horas".
83 case hardenedBiometry = "kand.bio"
84 /// Modo reforcado: a camada de fora que sobrevive a um rosto novo, `.biometryAny`. Em
85 /// "Continuar normal" e "Esperar 24 horas". Nunca embrulha a RK: so o miolo, que ainda
86 /// pede o PIN pela `kand.pin`.
87 case hardenedBiometryAny = "kand.qualquer"
88 /// Modo reforcado: a camada de dentro, `.applicationPassword` derivada do PIN. Duas
89 /// geracoes que se alternam na troca de PIN; o cabecalho do rk.and diz qual vale.
90 case hardenedPINA = "kand.pin.a"
91 case hardenedPINB = "kand.pin.b"
92
93 public var usesBiometry: Bool { [.biometry, .hardenedBiometry, .hardenedBiometryAny].contains(self) }
94 /// A chave morre quando o cadastro de rostos muda (`.biometryCurrentSet`).
95 public var followsEnrollment: Bool { self == .biometry || self == .hardenedBiometry }
96 public var usesPassword: Bool { [.pin, .pinPending, .hardenedPINA, .hardenedPINB].contains(self) }
97}

231 /// Um Face ID, e a vigia no mesmo contexto ja avaliado, agora sem interface: nenhum
232 /// segundo rosto. Se o contexto nao servir para a outra chave, ou ela recusar sem dizer
233 /// por que, a vigia volta "nao se sabe" e quem decide e o hash do cadastro. So a chave
234 /// que o chaveiro diz que nao existe e troca certa (conferir no aparelho, §2.6).
235 public func unwrap(_ blob: Data, slot: WrapSlot, reason: String, watching watch: Data, with watchSlot: WrapSlot) throws -> (SecureBytes, WatchResult) {
236 let context = Self.faceContext(reason: reason)
237 defer { context.invalidate() }
238 let opened = try decrypt(blob, slot: slot, context: context)
239 context.interactionNotAllowed = true
240 let query: [String: Any] = [
241 kSecClass as String: kSecClassKey,
242 kSecAttrApplicationTag as String: tag(watchSlot),
243 kSecAttrKeyType as String: kSecAttrKeyTypeECSECPrimeRandom,
244 kSecReturnRef as String: true,
245 kSecUseAuthenticationContext as String: context,
246 ]
247 var item: CFTypeRef?
248 let status = SecItemCopyMatching(query as CFDictionary, &item)
249 if status == errSecItemNotFound { return (opened, .gone) }
250 guard status == errSecSuccess, let item else { return (opened, .unknown) }
251 let key = item as! SecKey // swiftlint:disable:this force_cast
252 var error: Unmanaged<CFError>?
253 guard let plain = SecKeyCreateDecryptedData(key, Self.algorithm, watch as CFData, &error) else {
254 _ = error?.takeRetainedValue()
255 return (opened, .unknown)
256 }
257 return (opened, .opened(SecureBytes.consuming(plain)))
258 }

260 /// O contexto de um Face ID. Nenhum reuso de autenticacao: cada assinatura pede presenca
261 /// nova. Sem botao de alternativa do sistema: a alternativa e o PIN do app. Sem motivo,
262 /// nenhuma interface.
263 private static func faceContext(reason: String?) -> LAContext {
264 let context = LAContext()
265 context.touchIDAuthenticationAllowableReuseDuration = 0
266 context.localizedFallbackTitle = ""
267 if let reason { context.localizedReason = reason } else { context.interactionNotAllowed = true }
268 return context
269 }

271 private func decrypt(_ blob: Data, slot: WrapSlot, context: LAContext) throws -> SecureBytes {
272 let key: SecKey
273 do {
274 key = try privateKey(slot, context: context)
275 } catch {
276 throw EnclaveError.keyMissing
277 }
278 var error: Unmanaged<CFError>?
279 guard let plain = SecKeyCreateDecryptedData(key, Self.algorithm, blob as CFData, &error) else {
280 let nsError = (error?.takeRetainedValue() as Error?).map { $0 as NSError }
281 // Qualquer recusa da autenticacao (cancelar, ligacao chegando, bloqueio por
282 // tentativas, rosto nao reconhecido) e "nao desta vez", nao "a biometria
283 // mudou". Quem decide se o cadastro mudou e o RootKeyVault, pelo estado.
284 if nsError?.domain == LAErrorDomain || nsError?.code == Int(errSecUserCanceled) || nsError?.code == Int(errSecAuthFailed) {
285 throw EnclaveError.cancelled
286 }
287 throw EnclaveError.failed("desembrulho")
288 }
289 return SecureBytes.consuming(plain)
290 }

368extension SecureBytes {
369 /// Copia de um `CFData` que acabou de sair de uma API do sistema para o buffer
370 /// seguro, e apaga a origem. O `CFData` e imutavel pela API, mas a memoria e do
371 /// processo: zerar e o que encurta a janela do segredo em claro.
372 static func consuming(_ data: CFData) -> SecureBytes {
373 let length = CFDataGetLength(data)
374 let out = SecureBytes(capacity: max(length, 1))
375 if let pointer = CFDataGetBytePtr(data), length > 0 {
376 out.append(contentsOf: UnsafeBufferPointer(start: pointer, count: length))
377 memset_s(UnsafeMutableRawPointer(mutating: pointer), length, 0, length)
378 }
379 return out
380 }
381}
