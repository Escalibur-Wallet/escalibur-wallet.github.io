62simbolos_do_c() {
63 local plataforma="$1" ferramenta objetos
64 if [ "$plataforma" = mac ]; then
65 ferramenta=nm
66 objetos=$(find Kit/.build/debug/CArgon2.build Kit/.build/debug/CSecp256k1.build -name '*.o' 2>/dev/null | tr '\n' ' ')
67 else
68 ferramenta=$(nm_android); objetos=$(objetos_c_android "${2:-Release}")
69 fi
70 [ -n "$objetos" ] && [ -n "$ferramenta" ] || return 0
71 # shellcheck disable=SC2086
72 comm -23 <("$ferramenta" -u $objetos 2>/dev/null | awk 'NF && $NF !~ /:$/ {print $NF}' | sort -u) \
73 <("$ferramenta" -g --defined-only $objetos 2>/dev/null | awk 'NF >= 3 {print $NF}' | sort -u) \
74 | { [ "$plataforma" = mac ] && sed 's/^_//' || cat; } | sed "s/^/$plataforma /"
75}
76conferir_simbolos_do_c() {
77 local plataforma="$1" configuracao="${2:-Release}" atuais novos perigosos
78 atuais=$(simbolos_do_c "$plataforma" "$configuracao")
79 [ "$plataforma" = android ] && plataforma_dita="android ($configuracao)" || plataforma_dita="$plataforma"
80 if [ -z "$atuais" ]; then
81 aviso "checagem do C cega no $plataforma_dita: nenhum simbolo nos objetos do Argon2 e da secp256k1"
82 return
83 fi
84 perigosos=$(echo "$atuais" | awk '{print $2}' | nome_sem_fortificacao | grep -E "$c_proibidos" | sort -u | tr '\n' ' ')
85 # A chamada ao kernel sem passar por funcao nenhuma: a instrucao no desmonte dos objetos.
86 local desmonte objetos
87 if [ "$plataforma" = mac ]; then
88 objetos=$(find Kit/.build/debug/CArgon2.build Kit/.build/debug/CSecp256k1.build -name '*.o' 2>/dev/null | tr '\n' ' ')
89 # shellcheck disable=SC2086
90 desmonte=$(xcrun objdump -d --no-show-raw-insn $objetos 2>/dev/null)
91 else
92 objetos=$(objetos_c_android "$configuracao")
93 # shellcheck disable=SC2086
94 desmonte=$("$(objdump_android)" -d --no-show-raw-insn $objetos 2>/dev/null)
95 fi
96 if [ -z "$desmonte" ]; then
97 aviso "desmonte do C cego no $plataforma_dita"
98 elif echo "$desmonte" | grep -qE '[[:space:]](svc|hvc|smc|syscall|sysenter)([[:space:]]|$)'; then
99 perigosos="$perigosos chamada-direta-ao-kernel"
100 fi
101 novos=$(comm -23 <(echo "$atuais" | sort -u) <(grep "^$plataforma " simbolos-c.lock 2>/dev/null | sort -u) | awk '{print $2}' | tr '\n' ' ')
102 if [ -n "$perigosos" ]; then
103 aviso "o C do $plataforma_dita chama o que nunca pode: $perigosos"
104 elif [ -n "$novos" ]; then
105 aviso "simbolo de fora novo no C do $plataforma_dita: $novos(revise e rode --atualizar-simbolos)"
106 else
107 ok "o C do $plataforma_dita (Argon2, secp256k1 e a cola) so chama os $(echo "$atuais" | wc -l | tr -d ' ') simbolos de simbolos-c.lock"
108 fi
109}
